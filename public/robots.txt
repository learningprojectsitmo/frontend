# Лендинг открыт для индексации — это единственная публичная страница,
# которая имеет смысл искать.
#
# /privacy намеренно закрыт: документ помечен как черновик и содержит
# плейсхолдеры вместо реквизитов оператора («[указать наименование,
# ИНН/ОГРН и адрес]»). Пока он не дописан, в выдаче ему не место.
# Со страницы он по-прежнему доступен по прямой ссылке из футера.
#
# Глобального Disallow: / нет, поэтому публичные адреса индексируются по
# умолчанию. Закрыто всё, что за авторизацией, и /join: в /join?token=...
# лежит пригласительный токен, попадание такого URL в выдачу недопустимо.
# robots.txt не умеет фильтровать query-параметры надёжно, поэтому /join
# закрыт целиком, а не по параметру.
User-agent: *
Disallow: /privacy
Disallow: /app
Disallow: /auth
Disallow: /join
Disallow: /wiki
Disallow: /notifications

Sitemap: /sitemap.xml
